const GROQ_URL = "https://api.groq.com/openai/v1/chat/completions"; const CHAT_MODEL = "openai/gpt-oss-120b"; const IMAGE_MODEL = "@cf/black-forest-labs/flux-1-schnell"; const CORS_HEADERS = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Methods": "GET, POST, OPTIONS", "Access-Control-Allow-Headers": "Content-Type", }; function json(data, status = 200) { return new Response(JSON.stringify(data), { status, headers: { ...CORS_HEADERS, "Content-Type": "application/json; charset=utf-8", "Cache-Control": "no-store", }, }); } function errorMessage(error) { return error instanceof Error ? error.message : String(error); } export default { async fetch(request, env) { const url = new URL(request.url); if (request.method === "OPTIONS") { return new Response(null, { status: 204, headers: CORS_HEADERS }); } if (request.method === "GET" && url.pathname === "/") { return json({ name: "Nova Brain", status: "online", routes: ["/chat", "/image"], }); } if (request.method !== "POST") { return json({ error: "Use POST for this endpoint." }, 405); } let body; try { body = await request.json(); } catch { return json({ error: "Invalid JSON request body." }, 400); } // ---------------- CHAT ---------------- if (url.pathname === "/chat") { if (!env.GROQ_API_KEY) { return json({ error: "Missing GROQ_API_KEY secret in Cloudflare Worker settings.", }, 500); } const messages = body.messages; if (!Array.isArray(messages) || messages.length === 0) { return json({ error: "Send a messages array." }, 400); } // Only allow valid roles and text content from the conversation. const safeMessages = messages .filter(m => m && ["user", "assistant"].includes(m.role) && typeof m.content === "string" ) .slice(-20) .map(m => ({ role: m.role, content: m.content.slice(0, 12000), })); if (!safeMessages.length || safeMessages[safeMessages.length - 1].role !== "user") { return json({ error: "The last message must be a user message." }, 400); } try { const response = await fetch(GROQ_URL, { method: "POST", headers: { "Authorization": `Bearer ${env.GROQ_API_KEY}`, "Content-Type": "application/json", }, body: JSON.stringify({ model: CHAT_MODEL, messages: [ { role: "system", content: "You are Nova, a helpful, friendly personal AI assistant. " + "Be clear, practical, and creative. Explain things simply when appropriate. " + "Never claim to have performed an action you did not perform." }, ...safeMessages, ], temperature: 0.7, max_completion_tokens: 1800, }), }); const result = await response.json(); if (!response.ok) { console.error("Groq API error:", response.status, result); return json({ error: result?.error?.message || `Chat provider returned HTTP ${response.status}. Check your Groq model and API key.`, }, 502); } const reply = result?.choices?.[0]?.message?.content; if (typeof reply !== "string" || !reply.trim()) { console.error("Unexpected Groq response:", result); return json({ error: "The AI provider returned an empty reply." }, 502); } return json({ reply: reply.trim() }); } catch (error) { console.error("Chat request failed:", errorMessage(error)); return json({ error: "Chat request failed. Check the Worker logs." }, 502); } } // ---------------- IMAGE GENERATION ---------------- if (url.pathname === "/image") { // The Cloudflare dashboard must have an AI binding named AI. if (!env.AI || typeof env.AI.run !== "function") { return json({ error: "Cloudflare AI binding is missing. Add a Workers AI binding named AI.", }, 500); } const prompt = typeof body.prompt === "string" ? body.prompt.trim() : ""; if (!prompt) { return json({ error: "Enter a description for the image." }, 400); } if (prompt.length > 1500) { return json({ error: "Please keep the image prompt under 1500 characters." }, 400); } try { const result = await env.AI.run(IMAGE_MODEL, { prompt: prompt, num_steps: 4, }); // FLUX.1-schnell normally returns { image: "base64..." }. // Handle a few possible response formats without exposing secrets. let image = result?.image; if (!image && result instanceof ArrayBuffer) { image = arrayBufferToBase64(result); } if (!image && result instanceof Uint8Array) { image = arrayBufferToBase64( result.buffer.slice( result.byteOffset, result.byteOffset + result.byteLength ) ); } if (!image && result?.data instanceof Uint8Array) { image = arrayBufferToBase64( result.data.buffer.slice( result.data.byteOffset, result.data.byteOffset + result.data.byteLength ) ); } if (typeof image !== "string" || !image.length) { console.error( "Unexpected Workers AI image response format:", result ? Object.keys(result) : result ); return json({ error: "The image model returned no image. Check Workers AI availability and Worker logs.", }, 502); } // The frontend accepts a data URL, so return one consistently. const dataUrl = image.startsWith("data:image/") ? image : `data:image/png;base64,${image}`; return json({ image: dataUrl }); } catch (error) { console.error("Image generation failed:", errorMessage(error)); return json({ error: "Image generation failed. Check Cloudflare Workers AI access, " + "model availability, usage limits, and the Worker logs.", }, 502); } } return json({ error: "Unknown endpoint. Use /chat or /image.", }, 404); }, }; function arrayBufferToBase64(buffer) { const bytes = new Uint8Array(buffer); let binary = ""; const chunkSize = 0x8000; for (let i = 0; i < bytes.length; i += chunkSize) { binary += String.fromCharCode(...bytes.subarray(i, i + chunkSize)); } return btoa(binary); }
